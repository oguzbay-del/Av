Protokol:
protocol RetryPolicy {
    func nextRetryInterval(attempt: Int) -> TimeInterval
    func shouldRetry(error: Error, attempt: Int) -> Bool
}

Uygulama - ExponentialBackoffRetry:
class ExponentialBackoffRetry: RetryPolicy {
    let baseDelay: TimeInterval = 2.0      // 2 saniye
    let maxDelay: TimeInterval = 300.0     // 5 dakika
    let jitterFactor: Double = 0.1         // ±10%
    let maxAttempts: Int = 5
    
    func nextRetryInterval(attempt: Int) -> TimeInterval {
        // Exponential: 2s, 4s, 8s, 16s, 32s
        let exponential = min(
            baseDelay * pow(2, Double(attempt - 1)),
            maxDelay
        )
        
        // Jitter ekle: ±10%
        let jitter = exponential * jitterFactor * Double.random(in: -1...1)
        return exponential + jitter
    }
    
    func shouldRetry(error: Error, attempt: Int) -> Bool {
        // Network errors için retry yap
        if error is URLError { return attempt < maxAttempts }
        // Diğer errors: retry yapma
        return false
    }
}

JitterCalculator:
class JitterCalculator {
    func calculate(_ baseInterval: TimeInterval, factor: Double) -> TimeInterval {
        let jitter = baseInterval * factor * Double.random(in: -1...1)
        return baseInterval + jitter
    }
}

Kullanım (WeatherService):
async func fetchWithRetry() throws -> WeatherForecast {
    var lastError: Error?
    let retryPolicy = ExponentialBackoffRetry()
    
    for attempt in 1...retryPolicy.maxAttempts {
        do {
            return try await fetch()
        } catch {
            lastError = error
            if !retryPolicy.shouldRetry(error, attempt: attempt) {
                break
            }
            
            let delay = retryPolicy.nextRetryInterval(attempt: attempt)
            try await Task.sleep(nanoseconds: UInt64(delay * 1_000_000_000))
        }
    }
    
    throw lastError ?? NetworkError.unknown
}

Retry Zaman Çizelgesi:
Attempt 1: Hemen (0s)
Attempt 2: 2s sonra
Attempt 3: 4s sonra
Attempt 4: 8s sonra
Attempt 5: 16s sonra

class AppLogger {
    static let shared = AppLogger()
    
    private var ringBuffer = RingBuffer<LogEntry>(capacity: 1000)
    private var fileWriter: FileWriter?
    private var consoleEnabled = DEBUG_BUILD
    
    struct LogEntry {
        let timestamp: Date
        let level: LogLevel
        let event: LogEvent
        let threadId: UInt64
    }
    
    // Ana logging fonksiyonu
    func log(_ event: LogEvent, level: LogLevel = .info) {
        let entry = LogEntry(
            timestamp: Date(),
            level: level,
            event: event,
            threadId: Thread.current.threadId
        )
        
        ringBuffer.append(entry)
        
        if consoleEnabled {
            printToConsole(entry)
        }
        
        fileWriter?.write(entry)
    }
    
    // Debug dosyası oluştur
    func dumpToFile() -> URL? {
        let fileName = "AvHaritasi_Log_\(Date().formatted()).txt"
        let documentsDir = FileManager.default.urls(
            for: .documentDirectory,
            in: .userDomainMask
        )[0]
        let fileURL = documentsDir.appendingPathComponent(fileName)
        
        var content = "AV HARITASI DEBUG LOG\n"
        content += "Oluşturulma: \(Date())\n"
        content += "====================\n\n"
        
        for entry in ringBuffer.allEntries {
            content += formatLogEntry(entry) + "\n"
        }
        
        try? content.write(to: fileURL, atomically: true, encoding: .utf8)
        return fileURL
    }
    
    // İstatistikler
    func metrics() -> LogMetrics {
        let entries = ringBuffer.allEntries
        return LogMetrics(
            totalEvents: entries.count,
            errorCount: entries.filter { $0.level == .error }.count,
            warningCount: entries.filter { $0.level == .warning }.count,
            timeSpan: entries.last.map { $0.timestamp.timeIntervalSince(entries.first?.timestamp ?? $0.timestamp) } ?? 0
        )
    }
    
    private func formatLogEntry(_ entry: LogEntry) -> String {
        let formatter = DateFormatter()
        formatter.dateFormat = "HH:mm:ss.SSS"
        let time = formatter.string(from: entry.timestamp)
        
        return "[\(time)] [\(entry.level)] \(entry.event)"
    }
}

--- ÇÖZÜM: Quadtree Veri Yapısı ---

Dosya: ios/AvHaritasi/Model/FastZoneQuadtree.swift

Nedir Quadtree?
Coğrafi veriler için space-partitioning ağacı.
- 2D düzlemi 4 bölüğe böler
- Her bölüğü özyinelemeli olarak böler
- Arama: O(log n) yerine O(n)

Kullanım:
Her zone:
- Merkezini hesapla
- Quadtree'ye ekle

Arama:
- Verilen koordinat için
- Quadtree'de hızlı bul
- 300m çapında sonuçlar döndür

Sınıf Yapısı:

class ZoneQuadtree {
    private struct QuadNode {
        let bounds: CGRect
        var zones: [ZoneClass] = []
        var children: [QuadNode]? = nil
        let maxZonesPerNode = 10
        let maxDepth = 8
    }
    
    private var root: QuadNode
    
    init(bounds: CGRect) {
        self.root = QuadNode(bounds: bounds)
    }
    
    // Zone ekle
    mutating func insert(_ zone: ZoneClass) {
        insert(zone, into: &root)
    }
    
    private mutating func insert(_ zone: ZoneClass, into node: inout QuadNode) {
        // Eğer zone node'un sınırları dışındaysa, atla
        guard node.bounds.contains(zone.center) else { return }
        
        // Eğer children yoksa ekle
        if node.children == nil {
            node.zones.append(zone)
            
            // Max zone sayısı aşıldıysa böl
            if node.zones.count > node.maxZonesPerNode {
                split(&node)
            }
        } else {
            // Children varsa, doğru quadrant'a ekle
            for i in 0..<4 {
                insert(zone, into: &node.children![i])
            }
        }
    }
    
    // Koordinat etrafında nearest zones bul
    func nearestZones(
        to coordinate: CLLocationCoordinate2D,
        within radius: CLLocationDistance
    ) -> [NearbyRestriction] {
        let cgPoint = CGPoint(
            x: coordinate.longitude,
            y: coordinate.latitude
        )
        let searchArea = CGRect(
            x: cgPoint.x - radius/111000,
            y: cgPoint.y - radius/111000,
            width: 2 * radius/111000,
            height: 2 * radius/111000
        )
        
        var results: [NearbyRestriction] = []
        query(searchArea, from: root, results: &results, center: cgPoint)
        
        // Mesafelerine göre sırala
        return results.sorted { $0.distance < $1.distance }
    }
    
    // Koordinatın bulunduğu zone
    func zoneAt(_ coordinate: CLLocationCoordinate2D) -> ZoneClass? {
        let cgPoint = CGPoint(
            x: coordinate.longitude,
            y: coordinate.latitude
        )
        return findZone(cgPoint, in: root)
    }
    
    // Tree'yi yeniden oluştur
    mutating func rebuild(with zones: [ZoneClass]) {
        // Sınırları hesapla
        var minLat = 90.0, maxLat = -90.0
        var minLon = 180.0, maxLon = -180.0
        
        for zone in zones {
            minLat = min(minLat, zone.center.latitude)
            maxLat = max(maxLat, zone.center.latitude)
            minLon = min(minLon, zone.center.longitude)
            maxLon = max(maxLon, zone.center.longitude)
        }
        
        let bounds = CGRect(
            x: minLon,
            y: minLat,
            width: maxLon - minLon,
            height: maxLat - minLat
        )
        
        self.root = QuadNode(bounds: bounds)
        
        for zone in zones {
            self.insert(zone)
        }
    }
}

nonisolated func locationManager(
    _ manager: CLLocationManager,
    didUpdateLocations locations: [CLLocation]
) {
    // Main thread'den hemen dön
    Task.detached(priority: .userInitiated) { [weak self] in
        for location in locations {
            await self?.processLocationAsync(location)
        }
    }
}

private func processLocationAsync(_ location: CLLocation) async {
    // 1. Eski konum kontrolleri (hafif)
    guard !demoActive else { return }
    guard location.horizontalAccuracy >= 0 else { return }
    guard abs(location.timestamp.timeIntervalSinceNow) < 30 else { return }
    
    // 2. Track'e ekle
    await MainActor.run {
        self.tracks.append(location)
    }
    
    // 3. Heavy computation (background'da)
    let assessment = await assessmentEngine.evaluate(
        location.coordinate,
        accuracy: location.horizontalAccuracy,
        at: AppClock.now(),
        context: context,
        settings: settings
    )
    
    // 4. Sonuçları main thread'e gönder
    await MainActor.run {
        self.location = location
        self.assessment = assessment
        self.locationError = nil
        
        // UI güncelleme (main thread'de)
        updatePowerMode(location)
        checkGeofence(location)
        notificationManager.alertIfNeeded(assessment)
        updateLiveStatus()
    }

// ZoneShapeView.swift (yeni veya güncelle)
struct ZoneShapeView: View {
    let zone: ZoneClass
    @State private var isRendered = false
    
    var body: some View {
        ZStack {
            if isRendered {
                // Render kapalı shape
                Shape {
                    path in
                    let cgPath = CGPath(
                        polygonPoints: zone.polygon
                    )
                    path.addPath(Path(cgPath))
                }
                .fill(zone.color)
                .opacity(0.6)
                .stroke(Color.black, lineWidth: 1)
            }
        }
        .onAppear {
            // Lazy render (birkaç frame delay)
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.1) {
                withAnimation(.easeIn(duration: 0.2)) {
                    isRendered = true
                }
            }
        }
    }
}

// MapView.swift güncellemesi
struct MapView: View {
    @State private var lastRenderTime: Date = .distantPast
    private let renderThrottle: TimeInterval = 0.1
    
    var body: some View {
        ZStack {
            // Harita base layers
            Map(...)
            
            // Zone overlays (throttled)
            ForEach(visibleZones, id: \.id) { zone in
                ZoneShapeView(zone: zone)
            }
        }
        .throttle(for: .milliseconds(100), scheduler: DispatchQueue.main)
    }
    
    // Viewport içinde görünen zones
    var visibleZones: [ZoneClass] {
        zoneQuadtree.nearestZones(
            to: mapCenter,
            within: mapRadius * 1.5
        ).map { $0.zone }
    }
}

--- KONTROL ---

4. Hedefi kontrol et

Sık Memory Kaçakları:
1. Closure'larda strong reference [weak self] kullan
2. Timer'ı invalidate et (deinit'de)
3. Delegation memory leak'leri
4. Image cache'leri temizle

--- ÇÖZÜM: WeakReference Kullan ---

// WRONG:
timer = Timer.scheduledTimer(withTimeInterval: 15, repeats: true) { _ in
    self.updateUI()
}

// RIGHT:
timer = Timer.scheduledTimer(withTimeInterval: 15, repeats: true) { [weak self] _ in
    self?.updateUI()
}
